# The test pack's scripted run (as scripted-run-core.txt; SIFTEC's suite's until 2026-11-12), with the popular-mods
# test pack loaded: Create's machines as moving contraptions, which lean hardest on entity and block-entity logic. Compared with vanilla's run by the world diff.
area minecraft:overworld -4 -4 3 3

cmd time set 1000
cmd weather clear 100000
cmd gamerule spawn_mobs false
# no random ticks: they draw on the level's shared random over every loaded chunk (cave vines, vines, crops), and
# vanilla's own runs differed in hundreds of blocks with them on; random-tick behaviour gets its own tests later
cmd gamerule random_tick_speed 0

# (the test pack's base world, Terralith's terrain, is below y 199 here: nothing to clear, run 43106805198)
cmd fill 0 199 0 31 199 31 minecraft:stone
# a moss carpet in the cave under 58,2,41 pops off at tick 0 (a pending tick from generation), and its item's
# velocity came from a random draw that differs between vanilla's own runs (vanilla a against b, run 43136722450's CI
# ITEMS lines), so the item ended up to 2 blocks away; the cave's moss carpets are taken out before tick 0
cmd fill 48 -6 32 68 12 50 minecraft:air replace minecraft:moss_carpet

# a mechanical bearing turning a small frame of planks (a rotating contraption); its creative motor is placed at tick
# 1 (below): Create ticks kinetics through its own hooks even while the game is frozen before tick 0, so a motor
# placed in the setup turned the bearing for however many frozen ticks the run waited for light (P2-1)
cmd setblock 2 201 2 create:mechanical_bearing[facing=up]
cmd setblock 2 202 2 minecraft:oak_planks
cmd setblock 1 202 2 minecraft:oak_planks
cmd setblock 3 202 2 minecraft:oak_planks
cmd setblock 2 202 1 minecraft:oak_planks
cmd setblock 2 202 3 minecraft:oak_planks

# a mechanical piston turned by a creative motor, pushing a block (a moving contraption)
cmd setblock 8 200 2 create:piston_extension_pole[facing=up]
cmd setblock 8 201 2 create:piston_extension_pole[facing=up]
cmd setblock 8 202 2 create:mechanical_piston[facing=up]
cmd setblock 8 203 2 minecraft:stone
# a motor on two sides: whichever horizontal axis the piston's shaft has, one of them turns it
cmd setblock 7 202 2 create:creative_motor[facing=east]
cmd setblock 8 202 1 create:creative_motor[facing=south]

# a hopper chain and a furnace, with the pack loaded (the furnace is placed at tick 1, below: with SIFTEC's pack it
# ticked through SIFTEC's hooks while the game was frozen, so one placed in the setup started cooking at a different
# moment in each run; P2-1. Kept on tick 1 with the test pack)
cmd setblock 2 201 8 minecraft:chest{Items:[{Slot:0b,id:"minecraft:cobblestone",count:64}]}
cmd fill 2 200 8 6 200 8 minecraft:hopper[facing=east]
cmd setblock 7 200 8 minecraft:chest

# a Create train (ci/server/harness/.../CiTrain.java): straight track, two stations beside it; the harness names
# the stations, builds and assembles a one-bogey carriage behind station A, and gives it a cycling schedule A to B
cmd fill 3 200 28 29 200 28 create:track[shape=xo]
cmd setblock 8 200 29 create:track_station{Id:[I;0,0,0,1],TargetTrack:[I;0,0,-1],TargetDirection:1b,Ortho:1b}
# station B faces the other way: a station takes trains from one direction only, and the train leaves A eastwards
cmd setblock 24 200 29 create:track_station{Id:[I;0,0,0,2],TargetTrack:[I;0,0,-1],TargetDirection:0b,Ortho:1b}

# animals and a villager
cmd summon minecraft:cow 14 200 14
cmd summon minecraft:sheep 16 200 14
cmd summon minecraft:villager 18 200 18

ticks 1200

# the bearing's motor and the furnace start on a fixed tick in every run (P2-1)
at 1 setblock 2 200 2 create:creative_motor[facing=up]
at 1 setblock 2 200 11 minecraft:furnace{Items:[{Slot:0b,id:"minecraft:raw_iron",count:8},{Slot:1b,id:"minecraft:coal",count:2}]}

# P2-1 diagnosis: the furnace and the bearing logged during the run
report 2 200 11
report 2 201 2

# P2-2: contraptions assembling and disassembling (ci/server/harness/.../CiContraption.java). The bearing (which takes
# the plank on top of it; the four beside that plank aren't attached and stay) assembles
# when its motor starts (tick 1); with the motor taken away it stops and places its frame back (its default mode),
# and assembles again when the motor returns. (Not the piston: its motors run from the setup, while the game waits
# frozen for a different number of ticks in each run, so its state at a given tick isn't a fixed point.)
at 300 ci-contraption check bearing assembled 2 202 2 2 202 2
at 400 setblock 2 200 2 minecraft:air
at 500 ci-contraption check bearing placed 2 202 2 2 202 2
at 600 setblock 2 200 2 create:creative_motor[facing=up]
at 700 ci-contraption check bearing assembled 2 202 2 2 202 2

at 20 ci-train stations 8 200 29 24 200 29
at 40 ci-train build
at 45 ci-train assemble
at 60 ci-train schedule

qs-at 0 quicksilver bench 50
