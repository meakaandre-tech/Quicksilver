# The core suite's random-tick run (P2-1): random ticks, precipitation and natural spawning, which the scripted run
# (scripted-run-core.txt) turns off. Played on the same base world and compared with vanilla's runs by the world diff,
# like the scripted run.
#   random-ticks MIN_Y   random ticks on, pinned per chunk (ci/server/harness/.../Pins.java): each chunk's random ticks
#                        and precipitation draw on a random seeded by the chunk and the game time, not the level's
#                        shared random, and only sections at or above MIN_Y tick (the scenes below; the terrain stays
#                        still, so its cave vines and grass can't move the shared random by how far the light engine's
#                        thread had got)
#   spawn-player X Y Z   a fake player in the overworld's player list only (the spawner needs one near: a mob farther
#                        than 128 blocks from every player isn't spawned), creative
#   spawn CX CZ          natural spawning in chunk CX CZ every tick (monsters; creatures every 400 game ticks), as the
#                        game spawns near a player, from a random pinned per chunk and tick
area minecraft:overworld -4 -4 3 3

cmd time set 1000
cmd weather rain 100000
# the harness spawns in the listed chunks itself; the game's own spawning (which needs a real player) stays off
cmd gamerule spawn_mobs false
cmd gamerule random_tick_speed 20
random-ticks 192
spawn-player 16.5 64 16.5
spawn -3 2
spawn 2 -3

# a two-layer stone platform high above the terrain: every scene below is at y 198 to 205 (sections 192 to 207)
cmd fill 0 198 0 31 199 31 minecraft:stone
